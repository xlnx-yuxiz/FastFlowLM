cmake_minimum_required(VERSION 3.20)
project(flm_chat_output_tests LANGUAGES CXX)
enable_testing()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(FLM_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(JSON_INCLUDE_DIR "${FLM_SOURCE}/include" CACHE PATH "Directory containing nlohmann/json.hpp")
add_library(flm_chat_parser STATIC)
foreach(name chat_args chat_format chat_msg chat_parser chat_stream chat_output)
    target_sources(flm_chat_parser PRIVATE "${FLM_SOURCE}/common/chat/${name}.cpp")
endforeach()
target_include_directories(flm_chat_parser PUBLIC "${FLM_SOURCE}/include" "${JSON_INCLUDE_DIR}")
if(MSVC)
    target_compile_options(flm_chat_parser PUBLIC /utf-8)
endif()
foreach(name chat_parser chat_output)
    add_executable(test_${name} test_${name}.cpp)
    target_link_libraries(test_${name} PRIVATE flm_chat_parser)
    add_test(NAME ${name} COMMAND test_${name})
endforeach()
